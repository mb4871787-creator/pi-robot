# ── System library notes ──────────────────────────────────────────────────────

#   sudo apt-get update
#   sudo apt-get install -y python3-opencv libzbar0
#
# For camera support:
#   sudo apt-get install -y python3-picamera2   (Pi Camera v2/3)
#   # OR use USB webcam with device_index: 0
#
# Enable UART on Raspberry Pi:
#   sudo raspi-config → Interface Options → Serial Port
#   Disable login shell, Enable serial port hardware
#   Add user to dialout group: sudo usermod -a -G dialout $USER
